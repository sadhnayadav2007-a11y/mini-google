DYNAMIC ALGORITHMS (PageRank + Keyword Frequency) // ============================================================================ /** * Computes iterative PageRank dynamically until convergence. * Formula: PR(A) = (1 - d) / N + d * Σ(PR(Ti) / C(Ti)) */ function calculatePageRank(pages, damping = 0.85, tolerance = 1e-9, maxIterations = 200) { const N = pages.length; } }); }); let iterations = 0; let maxDiff = Infinity; while (maxDiff > tolerance && iterations < maxIterations) { const nextRanks = {}; maxDiff = 0; // Account for any dangling nodes (though none exist in this graph) let danglingSum = 0; pages.forEach((p) => { if (outgoingCount[p.id] === 0) { danglingSum += ranks[p.id]; } }); pages.forEach((page) => { let incomingContribution = 0; incomingMap[page.id].forEach((linkerId) => { incomingContribution += ranks[linkerId] / outgoingCount[linkerId]; }); const newRank = (1 - damping) / N + damping * (incomingContribution + danglingSum / N); nextRanks[page.id] = newRank; const diff = Math.abs(newRank - ranks[page.id]); if (diff > maxDiff) maxDiff = diff; }); ranks = nextRanks; iterations += 1; } // Normalize so percentages sum cleanly to 100% const totalRank = Object.values(ranks).reduce((acc, v) => acc + v, 0); const normalized = {}; Object.keys(ranks).forEach((id) => { normalized[id] = ranks[id] / totalRank;
